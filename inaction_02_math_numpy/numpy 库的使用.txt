-------------------------------------------- 安装环境和 使用工具 ---------------------------------
# 在 /Users/yangjianwei/PyCharmMiscProject 目录下执行命令
pip install jupyter
pip install numpy

# 启动 jupyter
在 /Users/yangjianwei/PyCharmMiscProject 目录下 执行命令
jupyter notebook

-------------------------------------------- 安装相关的插件 ------------------------------------
# 安装 jupyter 相关的插件 注意: 最新版的不能使用, 因为最新版官方没有提供相应的扩展插件
pip install jupyter_contrib_nbextensions
pip install jupyter_nbextensions_configurator

# jupyter 开启的时候执行如下命令; 安装完成后 重新启动
jupyter contrib nbextension install --user
jupyter nbextensions_configurator enable --user
